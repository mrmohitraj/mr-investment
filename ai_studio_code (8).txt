import React, { useState, useRef } from 'react';
import { Product, Investment } from '../types';
import { 
  Lock, 
  Upload, 
  Plus, 
  Edit3, 
  Trash2, 
  Coins, 
  Image as ImageIcon, 
  Save, 
  X, 
  TrendingUp,
  RefreshCw,
  ShieldCheck
} from 'lucide-react';

interface AdminPanelProps {
  products: Product[];
  onUpdateProducts: (newProducts: Product[]) => void;
  investments: Investment[];
  onUpdateInvestments: (newInvestments: Investment[]) => void;
  onResetProductsToDefault: () => void;
}

export const AdminPanel: React.FC<AdminPanelProps> = ({
  products,
  onUpdateProducts,
  investments,
  onUpdateInvestments,
  onResetProductsToDefault,
}) => {
  const ADMIN_PIN = '391139';

  const [pinInput, setPinInput] = useState('');
  const [isAuthenticated, setIsAuthenticated] = useState(false);
  const [activeSubTab, setActiveSubTab] = useState<'coins' | 'orders'>('coins');

  const [editingCoin, setEditingCoin] = useState<Product | null>(null);
  const [isAddingNewCoin, setIsAddingNewCoin] = useState(false);

  const [coinName, setCoinName] = useState('');
  const [coinTag, setCoinTag] = useState('');
  const [coinPrice, setCoinPrice] = useState('');
  const [coinMinAmount, setCoinMinAmount] = useState<number>(10);
  const [coinReturnRate, setCoinReturnRate] = useState('');
  const [coinRisk, setCoinRisk] = useState<'Low' | 'Medium' | 'High Risk'>('Low');
  const [coinDescription, setCoinDescription] = useState('');
  const [coinLockPeriod, setCoinLockPeriod] = useState('Flexible');
  const [coinImage, setCoinImage] = useState('');
  const [imagePreview, setImagePreview] = useState('');

  const fileInputRef = useRef<HTMLInputElement>(null);

  const [editingInvId, setEditingInvId] = useState<string | null>(null);
  const [editStatus, setEditStatus] = useState<Investment['status']>('Order Booked');
  const [editDailyProfit, setEditDailyProfit] = useState<number>(0);
  const [editTotalReturn, setEditTotalReturn] = useState<number>(0);

  const handleAdminLogin = (e: React.FormEvent) => {
    e.preventDefault();
    if (pinInput.trim() === ADMIN_PIN) {
      setIsAuthenticated(true);
    } else {
      alert(`Galat Admin PIN! Correct PIN is: ${ADMIN_PIN}`);
    }
  };

  const handleFileUpload = (e: React.ChangeEvent<HTMLInputElement>) => {
    const file = e.target.files?.[0];
    if (file) {
      if (file.size > 5 * 1024 * 1024) {
        alert('File size exceeds 5MB limit. Please select a smaller icon image.');
        return;
      }
      const reader = new FileReader();
      reader.onloadend = () => {
        const result = reader.result as string;
        setCoinImage(result);
        setImagePreview(result);
      };
      reader.readAsDataURL(file);
    }
  };

  const openEditCoinModal = (prod: Product) => {
    setEditingCoin(prod);
    setIsAddingNewCoin(false);
    setCoinName(prod.name);
    setCoinTag(prod.tag);
    setCoinPrice(prod.price);
    setCoinMinAmount(prod.minAmount);
    setCoinReturnRate(prod.returnRate);
    setCoinRisk(prod.risk);
    setCoinDescription(prod.description);
    setCoinLockPeriod(prod.lockPeriod);
    setCoinImage(prod.image);
    setImagePreview(prod.image);
  };

  const openAddNewCoinModal = () => {
    setEditingCoin(null);
    setIsAddingNewCoin(true);
    setCoinName('');
    setCoinTag('New Coin');
    setCoinPrice('Min ₹100');
    setCoinMinAmount(100);
    setCoinReturnRate('20% Guaranteed');
    setCoinRisk('Low');
    setCoinDescription('Official investment token with high return potential.');
    setCoinLockPeriod('30 Days');
    setCoinImage('https://images.unsplash.com/photo-1621416894569-0f39ed31d247?auto=format&fit=crop&w=600&q=80');
    setImagePreview('https://images.unsplash.com/photo-1621416894569-0f39ed31d247?auto=format&fit=crop&w=600&q=80');
  };

  const handleSaveCoin = (e: React.FormEvent) => {
    e.preventDefault();
    if (!coinName.trim()) {
      alert('Coin Name is required');
      return;
    }

    if (isAddingNewCoin) {
      const newProduct: Product = {
        id: Date.now(),
        name: coinName.trim(),
        tag: coinTag.trim() || 'Custom Token',
        price: coinPrice.trim() || `Min ₹${coinMinAmount}`,
        minAmount: Number(coinMinAmount) || 10,
        image: coinImage || imagePreview,
        returnRate: coinReturnRate.trim() || 'Custom Returns',
        risk: coinRisk,
        description: coinDescription.trim() || 'Custom coin product',
        lockPeriod: coinLockPeriod.trim() || 'Flexible',
      };
      onUpdateProducts([...products, newProduct]);
    } else if (editingCoin) {
      const updatedList = products.map((p) => {
        if (p.id === editingCoin.id) {
          return {
            ...p,
            name: coinName.trim(),
            tag: coinTag.trim(),
            price: coinPrice.trim(),
            minAmount: Number(coinMinAmount),
            image: coinImage || imagePreview,
            returnRate: coinReturnRate.trim(),
            risk: coinRisk,
            description: coinDescription.trim(),
            lockPeriod: coinLockPeriod.trim(),
          };
        }
        return p;
      });
      onUpdateProducts(updatedList);
    }

    closeCoinModal();
  };

  const handleDeleteCoin = (coinId: number) => {
    if (confirm('Kya aap sach me is coin ko delete karna chahte hain?')) {
      const filtered = products.filter((p) => p.id !== coinId);
      onUpdateProducts(filtered);
    }
  };

  const closeCoinModal = () => {
    setEditingCoin(null);
    setIsAddingNewCoin(false);
  };

  const handleSaveOrderEdit = (invId: string) => {
    const updated = investments.map((inv) => {
      if (inv.id === invId) {
        return {
          ...inv,
          status: editStatus,
          dailyProfit: Number(editDailyProfit),
          totalReturn: Number(editTotalReturn),
        };
      }
      return inv;
    });
    onUpdateInvestments(updated);
    setEditingInvId(null);
  };

  const handleDeleteOrder = (invId: string) => {
    if (confirm('Kya aap is order record ko delete karna chahte hain?')) {
      onUpdateInvestments(investments.filter((inv) => inv.id !== invId));
    }
  };

  if (!isAuthenticated) {
    return (
      <div className="max-w-md mx-auto my-12 bg-slate-900/90 border border-amber-500/30 rounded-3xl p-8 space-y-6 shadow-2xl backdrop-blur-md">
        <div className="text-center space-y-2">
          <div className="w-16 h-16 rounded-2xl bg-gradient-to-tr from-amber-500 to-amber-400 text-slate-950 font-black text-2xl mx-auto flex items-center justify-center shadow-lg shadow-amber-500/20">
            <Lock className="w-8 h-8 text-slate-950" />
          </div>
          <h2 className="text-2xl font-black text-white">Admin Control Center</h2>
          <p className="text-xs text-amber-400 font-bold uppercase tracking-wider">Enter Admin PIN: 391139</p>
        </div>

        <form onSubmit={handleAdminLogin} className="space-y-4">
          <div>
            <label className="block text-slate-300 font-bold text-xs mb-1.5">Admin Security PIN</label>
            <input
              type="password"
              required
              placeholder="Enter PIN (391139)"
              value={pinInput}
              onChange={(e) => setPinInput(e.target.value)}
              className="w-full bg-slate-950 border border-slate-700 rounded-xl px-4 py-3 text-center text-white font-black text-lg focus:border-amber-500 focus:outline-none tracking-widest"
            />
          </div>

          <button
            type="submit"
            className="w-full py-3.5 rounded-xl bg-amber-500 hover:bg-amber-400 text-slate-950 font-black text-xs uppercase tracking-wider shadow-lg transition-all cursor-pointer"
          >
            Unlock Admin Panel
          </button>
        </form>

        <p className="text-[11px] text-slate-500 text-center">
          Admin code updated to <strong className="text-amber-400">391139</strong> for secure coin icon uploads and order management.
        </p>
      </div>
    );
  }

  return (
    <div className="space-y-6">
      <div className="bg-slate-900 border border-slate-800 rounded-3xl p-6 flex flex-col md:flex-row items-center justify-between gap-4">
        <div>
          <div className="flex items-center space-x-2">
            <ShieldCheck className="w-5 h-5 text-amber-400" />
            <h2 className="text-2xl font-black text-white">Admin Management Dashboard</h2>
          </div>
          <p className="text-xs text-slate-400 mt-1">
            Admin Code: <strong className="text-amber-400">391139</strong> | Custom Icon Uploads Enabled
          </p>
        </div>

        <div className="flex items-center space-x-2">
          <button
            onClick={() => setActiveSubTab('coins')}
            className={`px-4 py-2 rounded-xl text-xs font-bold transition-all cursor-pointer flex items-center space-x-1.5 ${
              activeSubTab === 'coins' ? 'bg-amber-500 text-slate-950 shadow-md' : 'bg-slate-800 text-slate-300 hover:text-white'
            }`}
          >
            <Coins className="w-4 h-4" />
            <span>Coin Icons & Products ({products.length})</span>
          </button>

          <button
            onClick={() => setActiveSubTab('orders')}
            className={`px-4 py-2 rounded-xl text-xs font-bold transition-all cursor-pointer flex items-center space-x-1.5 ${
              activeSubTab === 'orders' ? 'bg-amber-500 text-slate-950 shadow-md' : 'bg-slate-800 text-slate-300 hover:text-white'
            }`}
          >
            <TrendingUp className="w-4 h-4" />
            <span>Orders & Returns ({investments.length})</span>
          </button>
        </div>
      </div>

      {activeSubTab === 'coins' && (
        <div className="space-y-6">
          <div className="flex flex-col sm:flex-row sm:items-center justify-between gap-4 bg-slate-900/60 p-4 rounded-2xl border border-slate-800">
            <div>
              <h3 className="text-base font-bold text-white">Custom Coin Icon Management</h3>
              <p className="text-xs text-slate-400">
                Upload custom icon images (.png, .jpg, .svg) for any coin. Custom uploaded icons will be displayed live on the platform.
              </p>
            </div>

            <div className="flex items-center space-x-2">
              <button
                onClick={openAddNewCoinModal}
                className="px-4 py-2.5 rounded-xl bg-emerald-500 hover:bg-emerald-400 text-slate-950 font-black text-xs flex items-center space-x-1.5 transition-all cursor-pointer"
              >
                <Plus className="w-4 h-4" />
                <span>Add New Coin</span>
              </button>

              <button
                onClick={() => {
                  if (confirm('Aap saare coins ko default images ke saath reset karna chahte hain?')) {
                    onResetProductsToDefault();
                  }
                }}
                className="px-3 py-2.5 rounded-xl bg-slate-800 hover:bg-slate-700 text-slate-300 font-bold text-xs flex items-center space-x-1 transition-all cursor-pointer"
                title="Reset default coin images"
              >
                <RefreshCw className="w-3.5 h-3.5" />
                <span>Reset Defaults</span>
              </button>
            </div>
          </div>

          <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-3 gap-6">
            {products.map((prod) => (
              <div
                key={prod.id}
                className="bg-slate-900 border border-slate-800 rounded-3xl p-5 space-y-4 flex flex-col justify-between"
              >
                <div className="space-y-3">
                  <div className="relative w-full h-40 rounded-2xl overflow-hidden bg-slate-950 border border-slate-800 flex items-center justify-center p-2 group">
                    <img
                      src={prod.image}
                      alt={prod.name}
                      referrerPolicy="no-referrer"
                      className="w-full h-full object-cover rounded-xl"
                    />
                    <div className="absolute inset-0 bg-slate-950/70 opacity-0 group-hover:opacity-100 transition-opacity flex items-center justify-center space-x-2">
                      <button
                        onClick={() => openEditCoinModal(prod)}
                        className="px-3 py-1.5 rounded-xl bg-amber-500 text-slate-950 font-bold text-xs flex items-center space-x-1 cursor-pointer"
                      >
                        <Upload className="w-3.5 h-3.5" />
                        <span>Change Icon</span>
                      </button>
                    </div>
                  </div>

                  <div>
                    <span className="text-[10px] font-bold text-amber-400 bg-amber-500/10 px-2 py-0.5 rounded-full border border-amber-500/20">
                      {prod.tag}
                    </span>
                    <h4 className="text-base font-bold text-white mt-1">{prod.name}</h4>
                    <p className="text-xs text-emerald-400 font-bold">{prod.returnRate}</p>
                    <p className="text-[11px] text-slate-400 mt-1 line-clamp-2">{prod.description}</p>
                  </div>
                </div>

                <div className="pt-3 border-t border-slate-800/80 flex items-center justify-between">
                  <span className="text-sm font-black text-white">{prod.price}</span>
                  <div className="flex items-center space-x-1.5">
                    <button
                      onClick={() => openEditCoinModal(prod)}
                      className="p-2 rounded-xl bg-slate-800 hover:bg-slate-700 text-amber-400 transition-all cursor-pointer"
                      title="Edit coin details"
                    >
                      <Edit3 className="w-4 h-4" />
                    </button>
                    <button
                      onClick={() => handleDeleteCoin(prod.id)}
                      className="p-2 rounded-xl bg-slate-800 hover:bg-slate-700 text-rose-400 transition-all cursor-pointer"
                      title="Delete coin"
                    >
                      <Trash2 className="w-4 h-4" />
                    </button>
                  </div>
                </div>
              </div>
            ))}
          </div>
        </div>
      )}

      {activeSubTab === 'orders' && (
        <div className="bg-slate-900 border border-slate-800 rounded-3xl p-6 space-y-4">
          <h3 className="text-lg font-bold text-white">Investment Orders & Return Updates</h3>

          {investments.length === 0 ? (
            <div className="p-8 text-center text-slate-500 text-xs">
              Koi bhi investment order nahi mila hai. User ke dwara order place karte hi yahan show hoga.
            </div>
          ) : (
            <div className="space-y-4">
              {investments.map((inv) => (
                <div
                  key={inv.id}
                  className="bg-slate-950 p-5 rounded-2xl border border-slate-800 space-y-3"
                >
                  <div className="flex flex-col sm:flex-row sm:items-center justify-between gap-2 text-xs border-b border-slate-800/80 pb-3">
                    <div>
                      <span className="text-[10px] text-amber-400 font-bold uppercase">{inv.id}</span>
                      <h4 className="text-sm font-bold text-white">
                        User: {inv.userName} ({inv.userPhone})
                      </h4>
                      <p className="text-slate-400">Pincode: {inv.userPincode} | Date: {inv.date}</p>
                    </div>

                    <div className="text-left sm:text-right">
                      <div className="text-base font-black text-amber-400">₹{inv.amount} Deposit</div>
                      <div className="text-[11px] text-slate-400">UTR / Ref: <strong className="text-slate-200">{inv.utrNumber || 'N/A'}</strong></div>
                    </div>
                  </div>

                  <div className="flex flex-wrap items-center justify-between gap-3 text-xs">
                    <div>
                      <span className="text-slate-400 block text-[10px]">Purchased Coin:</span>
                      <span className="font-bold text-white">{inv.coinName}</span>
                    </div>

                    <div>
                      <span className="text-slate-400 block text-[10px]">Daily Profit Display:</span>
                      <span className="font-bold text-emerald-400">+₹{inv.totalReturn}</span>
                    </div>

                    <div>
                      <span className="text-slate-400 block text-[10px]">Current Status:</span>
                      <span
                        className={`px-2.5 py-1 rounded-lg text-[10px] font-bold ${
                          inv.status === 'Invest Success'
                            ? 'bg-emerald-500/20 text-emerald-400 border border-emerald-500/30'
                            : inv.status === 'Order Booked'
                            ? 'bg-blue-500/20 text-blue-400 border border-blue-500/30'
                            : inv.status === 'Rejected'
                            ? 'bg-rose-500/20 text-rose-400 border border-rose-500/30'
                            : 'bg-amber-500/20 text-amber-400 border border-amber-500/30'
                        }`}
                      >
                        {inv.status}
                      </span>
                    </div>
                  </div>

                  {editingInvId === inv.id ? (
                    <div className="bg-slate-900 p-4 rounded-xl border border-slate-800 space-y-3 mt-3">
                      <div className="grid grid-cols-1 sm:grid-cols-3 gap-3">
                        <div>
                          <label className="block text-slate-400 text-[10px] mb-1">Update Status</label>
                          <select
                            value={editStatus}
                            onChange={(e) => setEditStatus(e.target.value as Investment['status'])}
                            className="w-full bg-slate-950 border border-slate-700 rounded-lg p-2 text-white text-xs"
                          >
                            <option value="Order Book Pending">Order Book Pending</option>
                            <option value="Order Booked">Order Booked</option>
                            <option value="Invest Success">Invest Success</option>
                            <option value="Rejected">Rejected</option>
                          </select>
                        </div>

                        <div>
                          <label className="block text-slate-400 text-[10px] mb-1">Daily Profit Rate (₹)</label>
                          <input
                            type="number"
                            value={editDailyProfit}
                            onChange={(e) => setEditDailyProfit(Number(e.target.value))}
                            className="w-full bg-slate-950 border border-slate-700 rounded-lg p-2 text-white text-xs"
                          />
                        </div>

                        <div>
                          <label className="block text-slate-400 text-[10px] mb-1">Total Return Amount (₹)</label>
                          <input
                            type="number"
                            value={editTotalReturn}
                            onChange={(e) => setEditTotalReturn(Number(e.target.value))}
                            className="w-full bg-slate-950 border border-slate-700 rounded-lg p-2 text-white text-xs font-bold"
                          />
                        </div>
                      </div>

                      <div className="flex space-x-2">
                        <button
                          onClick={() => handleSaveOrderEdit(inv.id)}
                          className="px-4 py-2 bg-emerald-500 text-slate-950 font-black rounded-lg text-xs"
                        >
                          Save Return
                        </button>
                        <button
                          onClick={() => setEditingInvId(null)}
                          className="px-4 py-2 bg-slate-800 text-slate-300 rounded-lg text-xs"
                        >
                          Cancel
                        </button>
                      </div>
                    </div>
                  ) : (
                    <div className="flex items-center space-x-2 pt-2 border-t border-slate-800/60">
                      <button
                        onClick={() => {
                          setEditingInvId(inv.id);
                          setEditStatus(inv.status);
                          setEditDailyProfit(inv.dailyProfit || 0);
                          setEditTotalReturn(inv.totalReturn || 0);
                        }}
                        className="px-3 py-1.5 rounded-lg bg-slate-800 hover:bg-slate-700 text-amber-400 text-xs font-bold flex items-center space-x-1 cursor-pointer"
                      >
                        <Edit3 className="w-3.5 h-3.5" />
                        <span>Update Return & Status</span>
                      </button>

                      <button
                        onClick={() => handleDeleteOrder(inv.id)}
                        className="px-3 py-1.5 rounded-lg bg-slate-800 hover:bg-slate-700 text-rose-400 text-xs font-bold flex items-center space-x-1 cursor-pointer"
                      >
                        <Trash2 className="w-3.5 h-3.5" />
                        <span>Delete</span>
                      </button>
                    </div>
                  )}
                </div>
              ))}
            </div>
          )}
        </div>
      )}

      {(editingCoin || isAddingNewCoin) && (
        <div className="fixed inset-0 z-50 bg-slate-950/80 backdrop-blur-md flex items-center justify-center p-4 overflow-y-auto">
          <div className="bg-slate-900 border border-amber-500/30 rounded-3xl p-6 w-full max-w-xl space-y-4 shadow-2xl my-8">
            <div className="flex items-center justify-between border-b border-slate-800 pb-3">
              <h3 className="text-lg font-black text-white">
                {isAddingNewCoin ? 'Add New Coin with Custom Icon' : `Edit ${editingCoin?.name}`}
              </h3>
              <button
                onClick={closeCoinModal}
                className="p-1 rounded-lg bg-slate-800 text-slate-400 hover:text-white"
              >
                <X className="w-5 h-5" />
              </button>
            </div>

            <form onSubmit={handleSaveCoin} className="space-y-4 text-xs">
              <div className="bg-slate-950 p-4 rounded-2xl border border-slate-800 space-y-3">
                <label className="block text-slate-300 font-bold">Custom Coin Icon / Image File</label>
                
                <div className="flex flex-col sm:flex-row items-center gap-4">
                  <div className="w-24 h-24 rounded-2xl overflow-hidden bg-slate-900 border border-amber-500/30 flex items-center justify-center shrink-0">
                    {imagePreview ? (
                      <img src={imagePreview} alt="Preview" referrerPolicy="no-referrer" className="w-full h-full object-cover" />
                    ) : (
                      <ImageIcon className="w-8 h-8 text-slate-600" />
                    )}
                  </div>

                  <div className="space-y-2 flex-1 w-full">
                    <input
                      type="file"
                      ref={fileInputRef}
                      accept="image/*"
                      onChange={handleFileUpload}
                      className="hidden"
                    />
                    
                    <button
                      type="button"
                      onClick={() => fileInputRef.current?.click()}
                      className="w-full py-2.5 px-4 rounded-xl bg-amber-500 hover:bg-amber-400 text-slate-950 font-black text-xs flex items-center justify-center space-x-2 transition-all cursor-pointer"
                    >
                      <Upload className="w-4 h-4" />
                      <span>Upload Custom Icon File</span>
                    </button>

                    <p className="text-[10px] text-slate-400">
                      Or paste direct image URL below:
                    </p>

                    <input
                      type="text"
                      placeholder="https://..."
                      value={coinImage}
                      onChange={(e) => {
                        setCoinImage(e.target.value);
                        setImagePreview(e.target.value);
                      }}
                      className="w-full bg-slate-900 border border-slate-700 rounded-xl px-3 py-2 text-white text-xs focus:border-amber-500 focus:outline-none"
                    />
                  </div>
                </div>
              </div>

              <div className="grid grid-cols-1 sm:grid-cols-2 gap-3">
                <div>
                  <label className="block text-slate-300 font-bold mb-1">Coin Name</label>
                  <input
                    type="text"
                    required
                    value={coinName}
                    onChange={(e) => setCoinName(e.target.value)}
                    className="w-full bg-slate-950 border border-slate-700 rounded-xl p-2.5 text-white"
                  />
                </div>

                <div>
                  <label className="block text-slate-300 font-bold mb-1">Badge Tag</label>
                  <input
                    type="text"
                    value={coinTag}
                    onChange={(e) => setCoinTag(e.target.value)}
                    className="w-full bg-slate-950 border border-slate-700 rounded-xl p-2.5 text-white"
                  />
                </div>
              </div>

              <div className="grid grid-cols-1 sm:grid-cols-3 gap-3">
                <div>
                  <label className="block text-slate-300 font-bold mb-1">Price Display</label>
                  <input
                    type="text"
                    value={coinPrice}
                    onChange={(e) => setCoinPrice(e.target.value)}
                    className="w-full bg-slate-950 border border-slate-700 rounded-xl p-2.5 text-white"
                  />
                </div>

                <div>
                  <label className="block text-slate-300 font-bold mb-1">Min Deposit (₹)</label>
                  <input
                    type="number"
                    value={coinMinAmount}
                    onChange={(e) => setCoinMinAmount(Number(e.target.value))}
                    className="w-full bg-slate-950 border border-slate-700 rounded-xl p-2.5 text-white"
                  />
                </div>

                <div>
                  <label className="block text-slate-300 font-bold mb-1">Locking Period</label>
                  <input
                    type="text"
                    value={coinLockPeriod}
                    onChange={(e) => setCoinLockPeriod(e.target.value)}
                    className="w-full bg-slate-950 border border-slate-700 rounded-xl p-2.5 text-white"
                  />
                </div>
              </div>

              <div>
                <label className="block text-slate-300 font-bold mb-1">Return Rate / Growth</label>
                <input
                  type="text"
                  value={coinReturnRate}
                  onChange={(e) => setCoinReturnRate(e.target.value)}
                  className="w-full bg-slate-950 border border-slate-700 rounded-xl p-2.5 text-white"
                />
              </div>

              <div>
                <label className="block text-slate-300 font-bold mb-1">Description</label>
                <textarea
                  rows={2}
                  value={coinDescription}
                  onChange={(e) => setCoinDescription(e.target.value)}
                  className="w-full bg-slate-950 border border-slate-700 rounded-xl p-2.5 text-white"
                />
              </div>

              <div className="pt-2 flex space-x-3">
                <button
                  type="submit"
                  className="flex-1 py-3 bg-emerald-500 hover:bg-emerald-400 text-slate-950 font-black rounded-xl text-xs flex items-center justify-center space-x-1.5 transition-all cursor-pointer"
                >
                  <Save className="w-4 h-4" />
                  <span>Save Coin Icon & Details</span>
                </button>
                <button
                  type="button"
                  onClick={closeCoinModal}
                  className="py-3 px-5 bg-slate-800 text-slate-300 rounded-xl text-xs font-bold cursor-pointer"
                >
                  Cancel
                </button>
              </div>
            </form>
          </div>
        </div>
      )}
    </div>
  );
};
