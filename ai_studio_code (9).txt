import React, { useState, useEffect } from 'react';
import { Product, Investment, UserProfile } from './types';
import { initialProducts } from './data/initialProducts';
import { Header } from './components/Header';
import { CoinCard } from './components/CoinCard';
import { InvestModal } from './components/InvestModal';
import { AdminPanel } from './components/AdminPanel';
import { ContactSection } from './components/ContactSection';
import { 
  Sparkles, 
  Coins, 
  User, 
  Phone, 
  Lock, 
  Wallet, 
  ArrowUpRight,
  Search
} from 'lucide-react';

export default function App() {
  const [user, setUser] = useState<UserProfile>(() => {
    const saved = localStorage.getItem('mr_user');
    return saved ? JSON.parse(saved) : { name: '', phone: '', isLoggedIn: false };
  });

  const [loginPhone, setLoginPhone] = useState('');
  const [loginName, setLoginName] = useState('');
  const [loginPassword, setLoginPassword] = useState('');

  const [activeTab, setActiveTab] = useState<'home' | 'my-investments' | 'admin' | 'contact'>('home');

  const [products, setProducts] = useState<Product[]>(() => {
    const saved = localStorage.getItem('mr_products');
    if (saved) {
      try {
        const parsed = JSON.parse(saved);
        if (Array.isArray(parsed) && parsed.length > 0) return parsed;
      } catch (e) {
        console.error('Error loading products from localStorage', e);
      }
    }
    return initialProducts;
  });

  const [investments, setInvestments] = useState<Investment[]>(() => {
    const saved = localStorage.getItem('mr_investments');
    return saved ? JSON.parse(saved) : [];
  });

  const [selectedCoin, setSelectedCoin] = useState<Product | null>(null);

  const [searchQuery, setSearchQuery] = useState('');
  const [selectedTagFilter, setSelectedTagFilter] = useState<string>('All');

  useEffect(() => {
    localStorage.setItem('mr_user', JSON.stringify(user));
  }, [user]);

  useEffect(() => {
    localStorage.setItem('mr_products', JSON.stringify(products));
  }, [products]);

  useEffect(() => {
    localStorage.setItem('mr_investments', JSON.stringify(investments));
  }, [investments]);

  const handleLogin = (e: React.FormEvent) => {
    e.preventDefault();
    if (!loginName.trim() || !loginPhone.trim()) {
      alert('Kripya apna naam aur mobile number zaroor dalein');
      return;
    }
    setUser({ name: loginName.trim(), phone: loginPhone.trim(), isLoggedIn: true });
  };

  const handleLogout = () => {
    setUser({ name: '', phone: '', isLoggedIn: false });
  };

  const handleResetProducts = () => {
    setProducts(initialProducts);
    localStorage.setItem('mr_products', JSON.stringify(initialProducts));
  };

  const handlePlaceOrder = (orderData: {
    userName: string;
    userPhone: string;
    userPincode: string;
    amount: number;
    utrNumber: string;
  }) => {
    if (!selectedCoin) return;

    const newInv: Investment = {
      id: `INV-${Math.floor(100000 + Math.random() * 900000)}`,
      userId: user.phone || orderData.userPhone,
      userName: orderData.userName,
      userPhone: orderData.userPhone,
      userPincode: orderData.userPincode,
      coinName: selectedCoin.name,
      coinId: selectedCoin.id,
      amount: orderData.amount,
      date: new Date().toLocaleDateString('en-IN'),
      status: 'Order Book Pending',
      dailyProfit: 0,
      totalReturn: 0,
      utrNumber: orderData.utrNumber,
    };

    setInvestments([newInv, ...investments]);

    const message = `*NEW INVESTMENT ORDER - MR INVESTMENT*%0A%0A*Name:* ${encodeURIComponent(orderData.userName)}%0A*Mobile:* ${encodeURIComponent(orderData.userPhone)}%0A*Pincode:* ${encodeURIComponent(orderData.userPincode)}%0A*Coin Purchased:* ${encodeURIComponent(selectedCoin.name)}%0A*Amount:* ₹${orderData.amount}%0A*UTR / Ref No:* ${encodeURIComponent(orderData.utrNumber || 'Pending')}%0A*Status:* Order Book Pending`;
    window.open(`https://wa.me/916200873964?text=${message}`, '_blank');

    setSelectedCoin(null);
    setActiveTab('my-investments');
  };

  const userInvestments = investments.filter(
    (inv) => inv.userPhone === user.phone || inv.userId === user.phone
  );

  const totalUserInvested = userInvestments.reduce((acc, curr) => acc + curr.amount, 0);
  const totalUserReturn = userInvestments.reduce((acc, curr) => acc + (curr.totalReturn || 0), 0);

  const uniqueTags = ['All', ...Array.from(new Set(products.map((p) => p.tag)))];

  const filteredProducts = products.filter((p) => {
    const matchesSearch = p.name.toLowerCase().includes(searchQuery.toLowerCase()) ||
                          p.description.toLowerCase().includes(searchQuery.toLowerCase()) ||
                          p.tag.toLowerCase().includes(searchQuery.toLowerCase());
    const matchesTag = selectedTagFilter === 'All' || p.tag === selectedTagFilter;
    return matchesSearch && matchesTag;
  });

  return (
    <div className="min-h-screen bg-slate-950 text-slate-100 font-sans flex flex-col selection:bg-amber-500 selection:text-slate-950">
      <div className="bg-gradient-to-r from-amber-600 via-amber-500 to-emerald-600 text-slate-950 px-4 py-2 text-center text-xs font-black flex items-center justify-center space-x-2 shadow-md">
        <Sparkles className="w-4 h-4 shrink-0 animate-spin" />
        <span>Namaste sir, apka swagat hai hamare MR Investment website pe</span>
        <Sparkles className="w-4 h-4 shrink-0 animate-spin" />
      </div>

      {!user.isLoggedIn ? (
        <div className="flex-1 flex items-center justify-center p-4 my-8">
          <div className="w-full max-w-md bg-slate-900/90 border border-amber-500/30 rounded-3xl p-8 space-y-6 shadow-2xl backdrop-blur-md relative overflow-hidden">
            <div className="text-center space-y-2">
              <div className="w-16 h-16 rounded-2xl bg-gradient-to-tr from-amber-500 to-amber-400 text-slate-950 font-black text-2xl mx-auto flex items-center justify-center shadow-lg shadow-amber-500/20">
                MR
              </div>
              <h1 className="text-2xl font-black text-white tracking-tight">MR INVESTMENT</h1>
              <p className="text-xs text-amber-400 font-bold uppercase tracking-wider">Pahle Login Karein</p>
            </div>

            <form onSubmit={handleLogin} className="space-y-4 text-xs">
              <div>
                <label className="block text-slate-300 font-bold mb-1.5 flex items-center space-x-1">
                  <User className="w-3.5 h-3.5 text-amber-400" />
                  <span>Apka Naam (Full Name)</span>
                </label>
                <input
                  type="text"
                  required
                  placeholder="E.g. Ramesh Kumar"
                  value={loginName}
                  onChange={(e) => setLoginName(e.target.value)}
                  className="w-full bg-slate-950 border border-slate-700 rounded-xl px-4 py-3 text-white text-sm focus:border-amber-500 focus:outline-none transition-colors"
                />
              </div>

              <div>
                <label className="block text-slate-300 font-bold mb-1.5 flex items-center space-x-1">
                  <Phone className="w-3.5 h-3.5 text-amber-400" />
                  <span>Mobile Number</span>
                </label>
                <input
                  type="tel"
                  required
                  placeholder="10 digit phone number"
                  value={loginPhone}
                  onChange={(e) => setLoginPhone(e.target.value)}
                  className="w-full bg-slate-950 border border-slate-700 rounded-xl px-4 py-3 text-white text-sm focus:border-amber-500 focus:outline-none transition-colors"
                />
              </div>

              <div>
                <label className="block text-slate-300 font-bold mb-1.5 flex items-center space-x-1">
                  <Lock className="w-3.5 h-3.5 text-amber-400" />
                  <span>Password</span>
                </label>
                <input
                  type="password"
                  required
                  placeholder="••••••••"
                  value={loginPassword}
                  onChange={(e) => setLoginPassword(e.target.value)}
                  className="w-full bg-slate-950 border border-slate-700 rounded-xl px-4 py-3 text-white text-sm focus:border-amber-500 focus:outline-none transition-colors"
                />
              </div>

              <button
                type="submit"
                className="w-full py-3.5 rounded-xl bg-gradient-to-r from-amber-500 to-amber-600 hover:from-amber-400 hover:to-amber-500 text-slate-950 font-black text-sm uppercase tracking-wider shadow-lg shadow-amber-500/20 transition-all cursor-pointer"
              >
                Login & Enter MR Investment
              </button>
            </form>

            <div className="text-[11px] text-slate-500 text-center pt-2 border-t border-slate-800">
              Support & Admin Helpdesk: <strong className="text-amber-400">6200873964</strong>
            </div>
          </div>
        </div>
      ) : (
        <>
          <Header
            user={user}
            activeTab={activeTab}
            setActiveTab={setActiveTab}
            onLogout={handleLogout}
            investmentCount={userInvestments.length}
          />

          <main className="max-w-7xl mx-auto px-4 py-8 space-y-8 flex-1 w-full">
            {activeTab === 'home' && (
              <div className="space-y-6">
                <div className="bg-slate-900 border border-slate-800 rounded-3xl p-6 flex flex-col lg:flex-row items-center justify-between gap-6 shadow-xl">
                  <div className="space-y-1 text-center lg:text-left">
                    <h2 className="text-2xl font-black text-white tracking-tight flex items-center justify-center lg:justify-start space-x-2">
                      <span>Select Investment Coin</span>
                      <Coins className="w-6 h-6 text-amber-400" />
                    </h2>
                    <p className="text-xs text-slate-400">
                      PhonePe / GPay: <strong className="text-amber-400">6200873964</strong> | UPI ID: <strong className="text-emerald-400">6200873964@ybl</strong>
                    </p>
                  </div>

                  <div className="flex flex-col sm:flex-row items-center gap-3 w-full lg:w-auto">
                    <div className="relative w-full sm:w-64">
                      <Search className="w-4 h-4 text-slate-500 absolute left-3 top-3" />
                      <input
                        type="text"
                        placeholder="Search coins..."
                        value={searchQuery}
                        onChange={(e) => setSearchQuery(e.target.value)}
                        className="w-full bg-slate-950 border border-slate-700 rounded-xl pl-9 pr-4 py-2 text-xs text-white focus:border-amber-500 focus:outline-none"
                      />
                    </div>

                    <div className="flex items-center space-x-1 overflow-x-auto max-w-full pb-1 sm:pb-0">
                      {uniqueTags.map((tag) => (
                        <button
                          key={tag}
                          onClick={() => setSelectedTagFilter(tag)}
                          className={`px-3 py-1.5 rounded-xl text-[11px] font-bold whitespace-nowrap transition-all cursor-pointer ${
                            selectedTagFilter === tag
                              ? 'bg-amber-500 text-slate-950'
                              : 'bg-slate-950 text-slate-400 hover:text-white border border-slate-800'
                          }`}
                        >
                          {tag}
                        </button>
                      ))}
                    </div>
                  </div>
                </div>

                {filteredProducts.length === 0 ? (
                  <div className="bg-slate-900 border border-slate-800 rounded-3xl p-12 text-center text-slate-400 space-y-2">
                    <Coins className="w-10 h-10 text-slate-600 mx-auto" />
                    <p className="font-bold text-sm">Koi coin nahi mil raha hai.</p>
                    <p className="text-xs text-slate-500">Kripya search term badlein ya Admin Panel me naya coin icon upload karein.</p>
                  </div>
                ) : (
                  <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-3 gap-6">
                    {filteredProducts.map((prod) => (
                      <CoinCard
                        key={prod.id}
                        product={prod}
                        onInvest={(p) => setSelectedCoin(p)}
                      />
                    ))}
                  </div>
                )}
              </div>
            )}

            {activeTab === 'my-investments' && (
              <div className="space-y-6">
                <div className="grid grid-cols-1 sm:grid-cols-2 gap-6">
                  <div className="bg-slate-900 border border-slate-800 p-6 rounded-3xl shadow-xl space-y-2">
                    <span className="text-xs text-slate-400 font-bold uppercase tracking-wider block">Total Active Capital</span>
                    <div className="text-3xl font-black text-white">₹{totalUserInvested.toLocaleString('en-IN')}</div>
                    <span className="text-[11px] text-slate-500 block">Total coins purchased in current account</span>
                  </div>

                  <div className="bg-slate-900 border border-slate-800 p-6 rounded-3xl shadow-xl space-y-2">
                    <span className="text-xs text-slate-400 font-bold uppercase tracking-wider block">Daily Return Profit (Green Display)</span>
                    <div className="text-3xl font-black text-emerald-400 flex items-center space-x-1">
                      <span>+₹{totalUserReturn.toLocaleString('en-IN')}</span>
                      <ArrowUpRight className="w-6 h-6 text-emerald-400" />
                    </div>
                    <span className="text-[11px] text-emerald-500 block">Accumulated growth returns updated by admin</span>
                  </div>
                </div>

                <div className="bg-slate-900 border border-slate-800 rounded-3xl p-6 space-y-4 shadow-xl">
                  <h3 className="text-lg font-bold text-white flex items-center space-x-2">
                    <Wallet className="w-5 h-5 text-amber-400" />
                    <span>My Investment History</span>
                  </h3>

                  {userInvestments.length === 0 ? (
                    <div className="p-8 text-center text-slate-500 text-xs space-y-3">
                      <p>Aapke pas abhi tak koi active investment order nahi hai.</p>
                      <button
                        onClick={() => setActiveTab('home')}
                        className="px-4 py-2 bg-amber-500 text-slate-950 font-bold rounded-xl text-xs"
                      >
                        Explore Coins & Invest Now
                      </button>
                    </div>
                  ) : (
                    <div className="space-y-4">
                      {userInvestments.map((inv) => (
                        <div
                          key={inv.id}
                          className="bg-slate-950 p-5 rounded-2xl border border-slate-800 flex flex-col sm:flex-row justify-between items-start sm:items-center gap-4 text-xs"
                        >
                          <div className="space-y-1">
                            <span className="text-[10px] text-amber-400 font-bold uppercase">{inv.id}</span>
                            <h4 className="font-bold text-white text-base">{inv.coinName}</h4>
                            <div className="text-slate-400">
                              Amount: <strong className="text-white">₹{inv.amount}</strong> | Date: {inv.date} | Pincode: {inv.userPincode}
                            </div>
                            <div className="text-slate-500 text-[11px]">UTR: {inv.utrNumber || 'Pending'}</div>
                          </div>

                          <div className="text-left sm:text-right space-y-1">
                            <div className="font-bold text-emerald-400 text-sm">
                              +₹{inv.totalReturn || 0} Daily Profit
                            </div>
                            <span
                              className={`inline-block px-3 py-1 rounded-lg text-[10px] font-bold ${
                                inv.status === 'Invest Success'
                                  ? 'bg-emerald-500/20 text-emerald-400 border border-emerald-500/30'
                                  : inv.status === 'Order Booked'
                                  ? 'bg-blue-500/20 text-blue-400 border border-blue-500/30'
                                  : inv.status === 'Rejected'
                                  ? 'bg-rose-500/20 text-rose-400 border border-rose-500/30'
                                  : 'bg-amber-500/20 text-amber-400 border border-amber-500/30'
                              }`}
                            >
                              {inv.status}
                            </span>
                          </div>
                        </div>
                      ))}
                    </div>
                  )}
                </div>
              </div>
            )}

            {activeTab === 'admin' && (
              <AdminPanel
                products={products}
                onUpdateProducts={setProducts}
                investments={investments}
                onUpdateInvestments={setInvestments}
                onResetProductsToDefault={handleResetProducts}
              />
            )}

            {activeTab === 'contact' && <ContactSection />}
          </main>
        </>
      )}

      {selectedCoin && (
        <InvestModal
          product={selectedCoin}
          defaultUserName={user.name}
          defaultUserPhone={user.phone}
          onClose={() => setSelectedCoin(null)}
          onSubmitOrder={handlePlaceOrder}
        />
      )}
    </div>
  );
}
